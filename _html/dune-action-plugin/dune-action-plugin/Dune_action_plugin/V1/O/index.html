Sourceval (let+) : 'a t -> ('a -> 'b) -> 'b t
  let+ a = g in
  h

is equivalent to:

  map g ~f:(fun a -> h)
Sourceval (and+) : 'a t -> 'b t -> ('a * 'b) t
  let+ a1 = g1
  and+ a2 = g2 in
  h

is equivalent to:

  both g1 g2 |> map ~f:(fun (a1, a2) -> h)
