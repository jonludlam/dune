let from_char = function
  (* Should match [cword] definition *)
  | 'a'..'z' | 'A'..'Z' | '0'..'9' | '_' | '\170' | '\181' | '\186'
  | '\192'..'\214' | '\216'..'\246' | '\248'..'\255' ->
    letter
  | '\n' ->
    not_letter ++ newline
  | _ ->
    not_letter
